"""知识库 service 全功能脚本验证（绕开 pytest，直接调用）。"""
import io
import os
import sys
import tempfile

sys.path.insert(0, '/Users/qiuzhibin/project/simulation/backend')
os.environ['KB_ROOT'] = tempfile.mkdtemp()

from app.services import knowledge_service as kb

def check(name, cond):
    print(('PASS' if cond else 'FAIL'), '-', name, flush=True)
    if not cond:
        raise SystemExit(f'FAILED: {name}')

# 1. 多级文件夹
r = kb.create_folder('法规')
check('create folder 法规', r['ok'] and r['path'] == '法规')
r = kb.create_folder('国际', '法规')
check('create sub folder 法规/国际', r['ok'] and r['path'] == '法规/国际')
check('dup folder rejected', not kb.create_folder('国际', '法规')['ok'])
check('bad parent rejected', not kb.create_folder('x', '不存在')['ok'])

# 2. 上传 txt
r = kb.upload('说明.txt', '第一条说明内容。\n第二条说明内容。'.encode('utf-8'))
check('upload txt ok', r['ok'])
did = r['doc']['id']
check('txt content parsed', '第一条说明内容' in kb.doc_content(did)['content'])

# 3. 上传 docx
from docx import Document
doc = Document()
doc.add_paragraph('Word 文档第一段')
doc.add_paragraph('Word 文档第二段')
buf = io.BytesIO(); doc.save(buf)
r = kb.upload('报告.docx', buf.getvalue())
check('upload docx ok', r['ok'] and r['doc']['ext'] == '.docx')
content = kb.doc_content(r['doc']['id'])['content']
check('docx content parsed', 'Word 文档第一段' in content and 'Word 文档第二段' in content)

# 4. 上传 pdf（reportlab 生成含文本的真实 PDF）
from reportlab.pdfgen import canvas
buf = io.BytesIO()
cv = canvas.Canvas(buf)
cv.drawString(72, 150, 'PDF KB Content')
cv.save()
r = kb.upload('手册.pdf', buf.getvalue())
check('upload pdf ok', r['ok'] and r['doc']['ext'] == '.pdf')
check('pdf content parsed', 'PDF KB Content' in kb.doc_content(r['doc']['id'])['content'])

# 5. 上传到子文件夹
r = kb.upload('子文档.txt', '子文件夹内容。'.encode('utf-8'), '法规/国际')
check('upload to subfolder', r['ok'] and r['doc']['folder'] == '法规/国际')

# 6. 不支持格式
check('unsupported rejected', not kb.upload('a.py', b'x')['ok'])

# 7. 重命名文件夹（含其下文档）
r = kb.rename_folder('法规', '规范')
check('rename folder', r['ok'] and r['path'] == '规范')
doc = kb.get_doc(r_id := kb.upload('在法规下.txt', '内容。'.encode('utf-8'), '规范')['doc']['id'])
check('folder rename keeps docs', doc['folder'] == '规范')

# 8. 重命名文档
r = kb.rename_doc(did, '新名字')
check('rename doc', r['ok'] and r['doc']['title'] == '新名字')

# 9. 树结构与统计（不应含 raw 目录）
tree = kb.list_tree()
check('tree ok', tree['ok'])
check('stats docs >= 5', tree['stats']['docs'] >= 5)
root_docs = {d['title'] for d in tree['tree']['docs']}
check('root docs titles', '新名字' in root_docs and '报告' in root_docs and '手册' in root_docs)

# 10. 搜索
res = kb.search('PDF KB Content')
check('search pdf content', res['results'] and res['results'][0]['doc']['title'] == '手册')
res = kb.search('Word 文档')
check('search docx content', any(x['doc']['title'] == '报告' for x in res['results']))

# 11. 删除文档
r = kb.delete_doc(did)
check('delete doc', r['ok'] and kb.get_doc(did) is None)

# 12. 级联删除文件夹
res = kb.delete_folder('规范')
check('cascade delete folder', res['ok'] and res['removed_docs'] >= 2)
check('root folder removed', not os.path.exists(os.path.join(os.environ['KB_ROOT'], '规范')))

# 13. 禁止删除根
check('delete root forbidden', not kb.delete_folder('')['ok'] and not kb.delete_folder('/')['ok'])

print('ALL 20+ CHECKS PASSED', flush=True)
